% -----------------------------------------------------------------------------
%  v3 drafting macros -- REMOVE ALL THREE BEFORE SUBMISSION
% -----------------------------------------------------------------------------

% \dataref{...}  the *evidence of record* for the claim just made: which batch,
% which DA, at what protocol. v2's \srcnote points at CODE, which answers "how
% is it implemented"; \dataref points at DATA, which answers "where does this
% number come from". Every number in Chapters 6-7 carries one.
\newcommand{\dataref}[1]{\textcolor{gray}{\footnotesize\textsf{[data: #1]}}}

% \provisional{...}  a statement that is true of the data on disk today but sits
% below the strength the thesis wants, and will move when an owed run lands.
% The list of owed runs is DATASTATUS_20260910 section 8; each use names its own.
\newcommand{\provisional}[1]{\textcolor{orange}{\textbf{[PROVISIONAL:} #1\textbf{]}}}

% \guard{...}  the caveat sentence that must travel with a claim wherever it is
% quoted -- seed depth, protocol tier, a selection effect. Set in the running
% text rather than a footnote, because a caveat in a footnote gets dropped when
% the sentence is quoted.
\newcommand{\guard}[1]{\textcolor{violet}{\emph{#1}}}

% -----------------------------------------------------------------------------
%  BIBLIOGRAPHY -- the v3 half
% -----------------------------------------------------------------------------
%  bibliography.bib is inherited from v2 byte-for-byte and is registered by the
%  inherited preamble. v3's own entries live in a SEPARATE resource so that the
%  two drafts never append to the same file -- see the header of
%  bibliography_v3.bib for why that matters, and inherited/SYNC_STATE.json for
%  which v2 revision the inherited half is currently at.
%
%  \addbibresource is additive; biber reports a key defined in both resources
%  rather than silently shadowing it. Guarded by \ifstandalone for the same
%  reason v2 guards its own: in the merged template build, settings.tex line 68
%  already registers the bibliography, and registering a resource twice there
%  duplicates it.
\ifstandalone
  \addbibresource{bibliography_v3.bib}
\fi

% -----------------------------------------------------------------------------
%  \autoref NAMES THAT NOTHING ELSE DEFINES
% -----------------------------------------------------------------------------
%  Two gaps that only bite once Chapters 5-8 exist, both fixed here rather than
%  in a chapter, so no inherited file diverges from v2 over them.
%
%  1. THEOREM-LIKE ENVIRONMENTS. The inherited preamble declares definition /
%     proposition / theorem / remark, but hyperref only knows \autoref names for
%     the standard sectioning units. sec:setup:metrics states the thesis's
%     definition of improvement as a `definition' and five places \autoref it,
%     which without this is an undefined control sequence at build time.
%     \providecommand, not \newcommand: settings.tex may define them on merge.
\providecommand{\definitionautorefname}{Definition}
\providecommand{\propositionautorefname}{Proposition}
\providecommand{\theoremautorefname}{Theorem}
\providecommand{\remarkautorefname}{Remark}

%  2. APPENDICES. After \appendix the chapter counter still carries the chapter
%     autoref name, so \autoref{app:repro} renders "Chapter C" -- which is why v2
%     wrote that one cross-reference as `Appendix~\ref{...}' by hand and recorded
%     it as its single \ref exception. Redefining the name inside \appendix fixes
%     it for every appendix reference at once and lets \autoref be used
%     uniformly. The hand-written exception in Chapter 4 still typesets
%     correctly, so nothing there needs touching.
\providecommand{\chapterautorefname}{Chapter}
\let\fmpccorigappendix\appendix
\renewcommand{\appendix}{\fmpccorigappendix\def\chapterautorefname{Appendix}}

% -----------------------------------------------------------------------------
%  TEMPLATE PARITY: the capitalised \autoref names
% -----------------------------------------------------------------------------
%  Template_DONT_CHANGE/settings.tex:44-52 capitalises the \autoref names
%  through the babel language hook:
%      \addto\extrasamerican{\def\chapterautorefname{Chapter} ...}
%  The standalone branch never loads settings.tex, so a standalone build gets
%  hyperref's own defaults instead and renders "section 5.2" where the merged
%  template build renders "Section 5.2". With 133 \autoref calls in the draft
%  that is a visible difference between the two builds of the same source.
%
%  Mirrored here through the SAME hook, so it behaves identically and is not
%  overwritten when babel selects the language at \begin{document} -- a plain
%  \providecommand in the preamble would be. Guarded by \ifstandalone so that
%  the merged build takes settings.tex's copy and this never fires twice.
\ifstandalone
  \addto\extrasamerican{%
    \def\chapterautorefname{Chapter}%
    \def\sectionautorefname{Section}%
    \def\subsectionautorefname{Subsection}%
    \def\subsubsectionautorefname{Subsubsection}%
  }
\fi

% -----------------------------------------------------------------------------
%  \todofigure -- a figure that is PLANNED but does not exist yet
% -----------------------------------------------------------------------------
%  Distinct from \fmpccgraphic's fallback, which handles "the file should be
%  here and is not". This one means "nobody has made this yet", and it carries
%  the SPEC: what the panel must show, and where the asset has to come from.
%  Written into the draft rather than into a side file so the gap is visible on
%  every read of the chapter, and so whoever renders it does not have to guess.
%  Counted by tools/check.py; must reach zero before submission.
%
%  #1 (optional) box height, default 0.30\textwidth · #2 the specification
\newcommand{\todofigure}[2][0.30\textwidth]{%
  \fcolorbox{red}{white}{%
    \parbox[c][#1][c]{0.90\linewidth}{%
      \centering
      {\color{red}\textbf{[FIGURE TO BE MADE]}}\\[0.7em]
      {\footnotesize #2}%
    }%
  }%
}
