% -----------------------------------------------------------------------------
%  LONG DATA SECTIONS OF THE APPENDIX  (author, 2026-09-24: "for something pure data
%  ... maybe use a weblink then just leave the conclusion in appendix; clearly mark the
%  long sections in head and mark what will hiding from appendix")
% -----------------------------------------------------------------------------
%  Four sections of Appendix B are data records. Each keeps its reading (one paragraph)
%  in the appendix and holds its tables inside \ifappendixfull ... \else ... \fi:
%    \appendixfulltrue   (default) prints the tables -- nothing is hidden yet;
%    \appendixfullfalse  hides them, leaving the paragraph, the web-link \hole and alias
%                        labels so that every \autoref from Chapter 6 still resolves.
%  bundle/make_bundle.py --appendix-short builds with the switch off; tools/check.py reads
%  the switch and checks only the active branch.

\appendixfulltrue

% \longdata{...}  the banner at the head of a long data section: what is a record here and
% what would move to the web link. A drafting mark (teal), printed in the annotated build;
% the clean build prints nothing. Counted by tools/check.py; must be gone before submission.
\newcommand{\longdata}[1]{\par\noindent\textcolor{teal}{\textbf{[LONG DATA:} #1\textbf{]}}\par}
